global _start

len equ  14

section .bss
res_buf: resb 32

section .text
_start:
    ; Function prologue
    push ebp
    mov ebp, esp

    ; Setup loop to copy msg into res_buf
    lea esi, msg
    lea edi, res_buf
    mov ecx, len

copy_loop:

    stosb
    loop copy_loop

    mov ebx, 1
    lea ecx, res_buf
    mov edx, len
    int 0x80

    ; Exit syscall: eax=1, ebx=0
    leave

    xor ebx, ebx
    int 0x80
